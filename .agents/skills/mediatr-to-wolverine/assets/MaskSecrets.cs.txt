// Wolverine logs "Invocation of <message.ToString()> failed!" when a handler or validator throws.
// Any message carrying a secret as a plain string/property would write it in clear text. Mask it.

using System.Text;

// 1) Plain record: define PrintMembers (record ToString uses it).
public sealed record SetPasswordCommand(string ServerProfile, string Identifier, string NewPassword)
{
   private bool PrintMembers(StringBuilder builder)
   {
      builder.Append($"ServerProfile = {ServerProfile}, Identifier = {Identifier}, NewPassword = ***");
      return true;
   }
};

// 2) Record deriving from a non-sealed record DTO that has the secret: override the virtual PrintMembers
//    and list only safe members (verified: the base DTO's Password is no longer printed).
public sealed record CreateUserCommand(string ServerProfile) : CreateUserDto
{
   protected override bool PrintMembers(StringBuilder builder)
   {
      builder.Append($"ServerProfile = {ServerProfile}, UserName = {UserName}, Password = ***");
      return true;
   }
};
